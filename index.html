function loginWithGoogleProfile() { if (typeof firebase === 'undefined') { alert('Erro: Firebase não carregou.'); return; } const provider = new firebase.auth.GoogleAuthProvider(); firebase.auth().signInWithPopup(provider) .then(result => { const user = result.user; selectProfile(user.displayName, user.photoURL || 'https://upload.wikimedia.org/wikipedia/commons/0/0b/Netflix-avatar.png'); }) .catch(err => { // Isso vai te mostrar na hora se é erro de domínio ou pop-up bloqueado alert('Erro no login do Google: ' + err.message); console.error(err); }); }
